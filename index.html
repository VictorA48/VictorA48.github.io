// Contador de visitas corregido y a prueba de fallos async function cargarVisitas() { const contadorEl = document.getElementById("contadorVisitas"); const claveLocal = "utv_vistas_locales"; try { // Petición a servidor de conteo const res = await fetch("https://api.counterapi.dev/v1/utv_oficial_py/visitas/up"); const data = await res.json(); // Verificamos que sea un número válido if (data && typeof data.count === "number" && !isNaN(data.count)) { contadorEl.textContent = data.count.toLocaleString("es-PY") + " vistas"; localStorage.setItem(claveLocal, data.count); return; } throw new Error("Dato no numérico"); } catch (err) { // Si la red externa falla o no responde, usa y suma en el navegador del usuario let vistasGuardadas = parseInt(localStorage.getItem(claveLocal) || "1", 10); vistasGuardadas += 1; localStorage.setItem(claveLocal, vistasGuardadas); contadorEl.textContent = vistasGuardadas.toLocaleString("es-PY") + " vistas"; } } cargarVisitas();
